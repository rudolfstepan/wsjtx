cmake_minimum_required(VERSION 3.16)
project(pskreporter_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Qt5 REQUIRED COMPONENTS Widgets Network)
add_executable(pskreporter_checks pskreporter_checks.cpp)
target_link_libraries(pskreporter_checks PRIVATE Qt5::Widgets Qt5::Network)
enable_testing()
add_test(NAME pskreporter_checks COMMAND pskreporter_checks)
set_tests_properties(pskreporter_checks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../build-custom/wsjtx_config.h")
  add_executable(about_preview about_preview.cpp ../../widgets/about.cpp ../../revision_utils.cpp)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../build-custom/qrc_wsjtx.cpp")
    target_sources(about_preview PRIVATE ../../build-custom/qrc_wsjtx.cpp)
  endif()
  set_target_properties(about_preview PROPERTIES AUTOUIC ON)
  target_include_directories(about_preview PRIVATE ../../build-custom ../../)
  target_compile_options(about_preview PRIVATE -include wsjtx_config.h)
  target_compile_definitions(about_preview PRIVATE CMAKE_BUILD)
  target_link_libraries(about_preview PRIVATE Qt5::Widgets)
endif()
