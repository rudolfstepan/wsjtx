cmake_minimum_required(VERSION 3.16)
project(webcontrol_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Qt5 REQUIRED COMPONENTS Widgets Network)
qt5_add_resources(web_resources ../../web/web.qrc)
add_executable(webcontrol_checks webcontrol_checks.cpp ../../Network/WebControlServer.cpp ${web_resources})
target_link_libraries(webcontrol_checks PRIVATE Qt5::Widgets Qt5::Network)
enable_testing()
add_test(NAME webcontrol_checks COMMAND webcontrol_checks)
set_tests_properties(webcontrol_checks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
